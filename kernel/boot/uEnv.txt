# ghOSt U-Boot Environment
# Allwinner H700 / Anbernic RG35XXH

bootdelay=1
console=ttyS0,115200
fdtfile=sun50i-h700-anbernic-rg35xxh.dtb

# Root partition UUID populated by build script
rootdev=UUID=ROOT_UUID
rootfstype=ext4
rootwait=rootwait ro

# Kernel command line
bootargs=console=${console} root=${rootdev} rootfstype=${rootfstype} ${rootwait} quiet loglevel=3 noatime

# Load and boot
bootcmd=load mmc 0:1 ${kernel_addr_r} Image; load mmc 0:1 ${fdt_addr_r} ${fdtfile}; booti ${kernel_addr_r} - ${fdt_addr_r}

kernel_addr_r=0x40080000
fdt_addr_r=0x4FA00000
ramdisk_addr_r=0x4FE00000
